4. Realiza un programa que calcule el área y perímetro del cuadrado
    #include <iostream>
    using namespace std;

    int main() {
        float lado, area, perimetro;

        cout << "Introduce el valor del lado del cuadrado: ";
        cin >> lado;

        area = lado * lado;
        perimetro = 4 * lado;

        cout << "Area del cuadrado: " << area << "\n";
        cout << "Perimetro del cuadrado: " << perimetro  << "\n";

        return 0;
    }

5. Realiza un programa que calcule el área y perímetro del rectángulo
    #include <iostream>
    using namespace std;

    int main() {

        float base, altura, area, perimetro;

        cout << "Introduce la base del rectangulo: ";
        cin >> base;

        cout << "Introduce la altura del rectangulo: ";
        cin >> altura;

        area = base * altura;
        perimetro = 2 * (base + altura);

        cout << "Area del rectangulo: " << area << "\n";
        cout << "Perimetro del rectangulo: " << perimetro << "\n";

        return 0;
    }
