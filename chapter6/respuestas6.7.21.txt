2) Algoritmo que imprima todos los cuadrados perfectos que hay entre dos números. Nota que
los números introducidos deben ser positivos.

    #include <iostream>
    #include <cmath>

    using namespace std;

    int main() {
        int inicio, fin;

        cout << "Ingrese el primer numero positivo: ";
        cin >> inicio;

        cout << "Ingrese el segundo numero positivo: ";
        cin >> fin;

        if (inicio < 0 || fin < 0) {
            cout << "Error: los numeros deben ser positivos." << endl;
            return 0;
        }

        if (inicio > fin) {
            int aux = inicio;
            inicio = fin;
            fin = aux;
        }

        cout << "Cuadrados perfectos entre " << inicio << " y " << fin << ":" << endl;

        for (int i = inicio; i <= fin; i++) {
            int raiz = sqrt(i);

            if (raiz * raiz == i) {
                cout << i << " ";
            }
        }

        return 0;
    }

12) Realiza un programa que pida tres números al usuario. Estos números representan una fecha:
día, mes y año. Lo que debe hacer el programa es validar que el día, mes y año sean válidos.
Un año válido es aquel que es mayor o igual que cero y menor a 10,000; un mes válido es
mayor a cero y menor a 13; Y un día válido es mayor a cero y menor a 29, 30 ó 31, según
sea el caso . Considera que si es febrero, entonces máximo puede ser 29 días. El programa
termina cuando la fecha es válida. Mientras solicitar la fecha.

    #include <iostream>

    using namespace std;

    int main() {
        int dia, mes, anio;
        bool fechaValida = false;

        while (!fechaValida) {
            cout << "Ingrese el dia: ";
            cin >> dia;

            cout << "Ingrese el mes: ";
            cin >> mes;

            cout << "Ingrese el anio: ";
            cin >> anio;

            fechaValida = true;

            if (anio < 0 || anio >= 10000) {
                fechaValida = false;
            }

            if (mes < 1 || mes > 12) {
                fechaValida = false;
            }

            int diasMes;

            if (mes == 2) {
                diasMes = 29;
            }
            else if (mes == 4 || mes == 6 || mes == 9 || mes == 11) {
                diasMes = 30;
            }
            else {
                diasMes = 31;
            }

            if (dia < 1 || dia > diasMes) {
                fechaValida = false;
            }

            if (!fechaValida) {
                cout << "Fecha invalida. Intente nuevamente.\n\n";
            }
        }

        cout << "\nFecha valida." << endl;

        return 0;
    }

22) Se desea tener un menú de opciones que de acuerdo a la opción seleccionada realice dicha
    operación:
    (a) La suma de todos los números menores a un número n.
    (b) Calcular cuántos números primos hay entre 0 y n.
    (c) Imprimir una pirámide de tamaño n y decir cuántos * hay.
    (d) Salir
    - De acuerdo a las opciones de arriba introducir el número de la izquierda para saber la
    opción a realizar.
    - Después de realizar la opción, deberá imprimirse el menu y volver a dar la opción.
    - El programa termina cuando el usuario selecciona la opción 4

    #include <iostream>

    using namespace std;

    bool esPrimo(int num) {
        if (num < 2)
            return false;

        for (int i = 2; i < num; i++) {
            if (num % i == 0)
                return false;
        }

        return true;
    }

    int main() {
        int opcion;

        do {
            cout << "\n===== MENU =====" << endl;
            cout << "1. Suma de numeros menores a n" << endl;
            cout << "2. Cantidad de numeros primos entre 0 y n" << endl;
            cout << "3. Imprimir piramide" << endl;
            cout << "4. Salir" << endl;
            cout << "Opcion: ";
            cin >> opcion;

            if (opcion == 1) {
                int n, suma = 0;

                cout << "Ingrese n: ";
                cin >> n;

                for (int i = 0; i < n; i++) {
                    suma += i;
                }

                cout << "Suma = " << suma << endl;
            }
            else if (opcion == 2) {
                int n, contador = 0;

                cout << "Ingrese n: ";
                cin >> n;

                for (int i = 0; i <= n; i++) {
                    if (esPrimo(i))
                        contador++;
                }

                cout << "Cantidad de primos = " << contador << endl;
            }
            else if (opcion == 3) {
                int n;
                int totalAsteriscos = 0;

                cout << "Ingrese el tamanio de la piramide: ";
                cin >> n;

                for (int i = 1; i <= n; i++) {
                    for (int j = 1; j <= i; j++) {
                        cout << "*";
                        totalAsteriscos++;
                    }
                    cout << endl;
                }

                cout << "Total de * = " << totalAsteriscos << endl;
            }

        } while (opcion != 4);

        cout << "Programa finalizado." << endl;

        return 0;
    }
