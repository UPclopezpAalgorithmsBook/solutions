3) Realiza un algoritmo que pida 4 números y los ordene de menor a mayor y viceversa.

    #include <iostream>
    using namespace std;

    int main() {

        int nums[4];

        for(int i = 0; i < 4; i++){
            cin >> nums[i];
        }

        // Ordenar de mayor a menor
        for(int i = 0; i < 4; i++){

            for(int j = i + 1; j < 4; j++){

                if(nums[i] > nums[j]){

                    int temp = nums[i];
                    nums[i] = nums[j];
                    nums[j] = temp;

                }

            }

        }

        for(int i = 0; i < 4; i++){
            cout << nums[i] << " ";
        }

        cout << endl;
        
        for(int i = 3; i >= 0; i--){
            cout << nums[i] << " ";
        }
    }
